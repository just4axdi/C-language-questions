Question 1

Write a program to show an example of pre-increment.

Coding

#include <stdio.h>

void main()
{
    int a = 5;

    printf("Value of a = %d\n", a);
    printf("Pre-increment = %d", ++a);
}

Value of a = 5
Pre-increment = 6

Question 3

Write a program to enter 2 numbers from user, add the numbers and show the answer, again take 2 numbers 
from user, subtract the numbers and display the answer, again take 2 numbers from user and multiply the 
numbers and display the answer, again take 2 numbers from user, divide the numbers 
and display the answer, and again take 2 numbers from user, find the remainder and display the answer?

Coding

#include <stdio.h>

void main()
{
    int a, b;

    printf("Enter two numbers for addition: ");
    scanf("%d %d", &a, &b);
    printf("Addition = %d\n", a + b);

    printf("Enter two numbers for subtraction: ");
    scanf("%d %d", &a, &b);
    printf("Subtraction = %d\n", a - b);

    printf("Enter two numbers for multiplication: ");
    scanf("%d %d", &a, &b);
    printf("Multiplication = %d\n", a * b);

    printf("Enter two numbers for division: ");
    scanf("%d %d", &a, &b);
    printf("Division = %d\n", a / b);

    printf("Enter two numbers for remainder: ");
    scanf("%d %d", &a, &b);
    printf("Remainder = %d\n", a % b);
}

Output

Enter two numbers for addition: 10 5
Addition = 15

Enter two numbers for subtraction: 10 5
Subtraction = 5

Enter two numbers for multiplication: 10 5
Multiplication = 50

Enter two numbers for division: 10 5
Division = 2

Enter two numbers for remainder: 10 3
Remainder = 1

Question 5

Write a program to enter a decimal number from the user (float) and print the right-hand side of the decimal point?

Coding

void main()
{
    float num;
    int decimal;

}

Question 7

Write a program to enter marks of students and calculate the average of students' marks.

Coding

#include <stdio.h>

void main()
{
    float m1, m2, m3, avg;

    printf("Enter marks of three subjects: ");
    scanf("%f %f %f", &m1, &m2, &m3);

    avg = (m1 + m2 + m3) / 3;

    printf("Average marks = %.2f", avg);
}

Output

Enter marks of three subjects: 80 70 90
Average marks = 80.00

Question 8

Write a program to calculate simple interest using the formula SI = (P × R × T) / 100.

Coding

#include <stdio.h>

void main()
{
    float P, R, T, SI;

    printf("Enter principal amount: ");
    scanf("%f", &P);

    printf("Enter rate of interest: ");
    scanf("%f", &R);

    printf("Enter time: ");
    scanf("%f", &T);

    SI = (P * R * T) / 100;

    printf("Simple Interest = %.2f", SI);
}

Output

Enter principal amount: 10000
Enter rate of interest: 5
Enter time: 2
Simple Interest = 1000.00
