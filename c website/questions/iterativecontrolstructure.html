Question 1

Enter a number from the user and check it is a prime number or not.

Coding

#include 

void main()
{
    int n, i, count = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    for(i = 1; i <= n; i++)
    {
        if(n % i == 0)
        {
            count++;
        }
    }

    if(count == 2)
    {
        printf("Prime Number");
    }
    else
    {
        printf("Not a Prime Number");
    }
}

Output

Enter a number: 7
Prime Number

Question 21

Enter a number from the user and check whether it is an Armstrong number or not.

Coding

#include 

void main()
{
    int n, original, rem, digits = 0, i;
    int sum = 0, power;

    printf("Enter a number: ");
    scanf("%d", &n);

    original = n;

    if(n < 0)
    {
        printf("Not an Armstrong Number");
    }
    else
    {
        int temp = n;

        do
        {
            digits++;
            temp = temp / 10;
        }
        while(temp != 0);

        temp = n;

        while(temp != 0)
        {
            rem = temp % 10;
            power = 1;

            for(i = 1; i <= digits; i++)
            {
                power = power * rem;
            }

            sum = sum + power;
            temp = temp / 10;
        }

        if(sum == original)
        {
            printf("Armstrong Number");
        }
        else
        {
            printf("Not an Armstrong Number");
        }
    }
}

Output

Enter a number: 153
Armstrong Number
