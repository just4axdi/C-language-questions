Question 1

Enter four numbers from user and print them into a matrix.

Coding

#include 

void main()
{
    int i, j, a[2][2];

    printf("Enter 4 numbers:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Matrix is:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", a[i][j]);
        }
        printf("\n");
    }
}

Output

1
2
3
4
Matrix is:

3 4

Question 3

Enter two matrices from user and add them.

Coding

#include 

void main()
{
    int i, j, a[2][2], b[2][2], c[2][2];

    printf("Enter first matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    printf("Addition of matrices:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            c[i][j] = a[i][j] + b[i][j];
            printf("%d ", c[i][j]);
        }
        printf("\n");
    }
}

Enter second matrix:

Question 5

Enter two matrices from user and multiply them.

Coding

#include

void main()
{
    int i, j, a[2][2], b[2][2], c[2][2];

    printf("Enter first matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            c[i][j] = a[i][0] * b[0][j] + a[i][1] * b[1][j];
        }
    }

    printf("Multiplication of matrices:\n");

    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", c[i][j]);
        }
        printf("\n");
    }
}

Output

Enter first matrix:
1 2
3 4
Enter second matrix:
5 6
7 8
Multiplication of matrices:
19 22
43 50
