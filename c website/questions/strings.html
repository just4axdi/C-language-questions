Question 1

Write a C program to enter a string from the user and convert it into uppercase?

Coding

#include <stdio.h>

int main()
{
    char str[100];
    int i = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[i] != '\0')
    {
        if(str[i] >= 'a' && str[i] <= 'z')
        {
            str[i] = str[i] - 32;
        }
        i++;
    }

    printf("Uppercase: %s", str);

    return 0;
}

Output

Enter a string: hello world
Uppercase: HELLO WORLD

enter a string from user and covert into title case?

Coding

{
    char str[100];

    int newWord = 1;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    {
        if(str[i] == ' ' || str[i] == '\n' || str[i] == '\t')

            newWord = 1;
        }
        else
        {
            if(str[i] >= 'A' && str[i] <= 'Z')
            {
                str[i] = str[i] + 32;
            }

            if(newWord == 1)
            {
                if(str[i] >= 'a' && str[i] <= 'z')
                {
                    str[i] = str[i] - 32;
                }
                newWord = 0;
            }
        }

        i++;
    }

    printf("Title case: %s", str);

    return 0;
}

Title case: Hello World Programming

Question 17

Write a C program to check whether a string is a palindrome or not by reversing
the string.

Coding

#include <stdio.h>
#include <string.h>

void main()
{
    char str[100], rev[100];
    int i, j, len;

    printf("Enter a string: ");
    gets(str);

    len = strlen(str);
    j = 0;

    for(i = len - 1; i >= 0; i--)
    {
        rev[j] = str[i];
        j++;
    }

    rev[j] = '\0';

    printf("Reverse string = %s\n", rev);

    if(strcmp(str, rev) == 0)
    {
        printf("String is palindrome");
    }
    else
    {
        printf("String is not palindrome");
    }
}

Output

Enter a string: madam
Reverse string = madam
String is palindrome

Question 18

Write a C program to enter the data of a student consisting of roll number, name, class, and marks of
five subjects, and display the entered data along with total and percentage.

Coding

#include <stdio.h>

void main()
{
    int roll, i, total = 0;
    char name[50], className[20];
    float marks[5], percentage;

    printf("Enter roll number: ");
    scanf("%d", &roll);

    printf("Enter name: ");
    scanf(" %49[^\n]", name);

    printf("Enter class: ");
    scanf(" %19[^\n]", className);

    for(i = 0; i < 5; i++)
    {
        printf("Enter marks of subject %d: ", i + 1);
        scanf("%f", &marks[i]);
        total = total + marks[i];
    }

    percentage = total / 5.0;

    printf("\n--- Student Details ---\n");
    printf("Roll Number: %d\n", roll);
    printf("Name: %s\n", name);
    printf("Class: %s\n", className);

    for(i = 0; i < 5; i++)
    {
        printf("Subject %d Marks: %.2f\n",
               i + 1, marks[i]);
    }

    printf("Total Marks: %d\n", total);
    printf("Percentage: %.2f%%\n", percentage);
}
