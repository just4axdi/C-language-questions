Question 10

{

    for(i = 0; i < 5; i++)

Question 11

Enter five values of array and sort in ascending order.

Coding

#include 

void main()
{
    int i, j, a[5], temp;

    printf("Enter 5 array elements:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < 5; i++)
    {
        for(j = i + 1; j < 5; j++)
        {
            if(a[i] > a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    printf("Array in ascending order:\n");

    for(i = 0; i < 5; i++)
    {
        printf("%d\n", a[i]);
    }
}

Output

Enter 5 array elements:
50
20
40
10
30
Array in ascending order:
10
20
30
40
50
